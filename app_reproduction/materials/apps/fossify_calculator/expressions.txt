# Format: expression = expected result  # note

(2 + 3) × 4 = 20        # parentheses first
10 ÷ 4 = 2.5            # division yields decimals
2 ^ 10 = 1024           # power
50 + 10% = 55           # percent applies to the preceding operand
100 − 25% = 75
200 × 10% = 20
0.1 + 0.2 = 0.3         # decimal arithmetic (display may round)
−5 × −5 = 25            # negative multiplication
1000000 × 1000000 = 1000000000000  # large numbers may be shown in full when scientific notation is not the truncation semantics
