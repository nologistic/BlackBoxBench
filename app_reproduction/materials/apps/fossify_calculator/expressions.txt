# 表达式样例与期望结果（复现自查参照）
# 格式：表达式 = 期望结果  # 说明

2 + 3 × 4 = 14          # 乘法优先
(2 + 3) × 4 = 20        # 括号优先
10 ÷ 4 = 2.5            # 除法产生小数
2 ^ 10 = 1024           # 幂运算
50 + 10% = 55           # 百分号作用于前操作数
100 − 25% = 75
200 × 10% = 20
0.1 + 0.2 = 0.3         # 小数运算（显示层面可四舍五入）
−5 × −5 = 25            # 负数乘法
1000000 × 1000000 = 1000000000000  # 大数不用科学计数法截断语义时可展示全数
