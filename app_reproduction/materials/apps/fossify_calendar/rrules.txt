# RRULE 子集速查与展开示例
# 假设"今天"= 2026-09-07（周一）

FREQ=DAILY;INTERVAL=1
  → 每天 09-08、09-09、09-10 …

FREQ=DAILY;INTERVAL=2
  → 09-09、09-11、09-13 …（隔天）

FREQ=WEEKLY;BYDAY=MO,TU,WE,TH,FR
  → 工作日：09-07（一）、09-08（二）、09-09（三）、09-10（四）、09-11（五）、09-14（一）…

FREQ=WEEKLY;BYDAY=SA,SU
  → 周末：09-12（六）、09-13（日）、09-19（六）…

FREQ=WEEKLY;BYDAY=WE
  → 每周三：09-09、09-16、09-23 …

FREQ=MONTHLY;BYMONTHDAY=15
  → 每月 15 日：09-15、10-15、11-15 …

FREQ=MONTHLY;BYMONTHDAY=1;INTERVAL=3
  → 每 3 个月的 1 日：10-01、01-01、04-01 …

FREQ=YEARLY;BYMONTH=9;BYMONTHDAY=10
  → 每年 9 月 10 日：2026-09-10、2027-09-10 …

FREQ=DAILY;UNTIL=20260915
  → 每天，直到（含）2026-09-15 后停止

FREQ=WEEKLY;BYDAY=TU;COUNT=4
  → 从首个周二起共 4 次：09-08、09-15、09-22、09-29，之后停止
