# RRULE subset quick reference and expansion examples
# Assume "today" = 2026-09-07 (Monday)

FREQ=DAILY;INTERVAL=1
  → daily: 09-08, 09-09, 09-10 …

FREQ=DAILY;INTERVAL=2
  → 09-09, 09-11, 09-13 … (every other day)

FREQ=WEEKLY;BYDAY=MO,TU,WE,TH,FR
  → weekdays: 09-07 (Mon), 09-08 (Tue), 09-09 (Wed), 09-10 (Thu), 09-11 (Fri), 09-14 (Mon) …

FREQ=WEEKLY;BYDAY=SA,SU
  → weekends: 09-12 (Sat), 09-13 (Sun), 09-19 (Sat) …

FREQ=WEEKLY;BYDAY=WE
  → every Wednesday: 09-09, 09-16, 09-23 …

FREQ=MONTHLY;BYMONTHDAY=15
  → the 15th monthly: 09-15, 10-15, 11-15 …

FREQ=MONTHLY;BYMONTHDAY=1;INTERVAL=3
  → the 1st every 3 months: 10-01, 01-01, 04-01 …

FREQ=YEARLY;BYMONTH=9;BYMONTHDAY=10
  → September 10 yearly: 2026-09-10, 2027-09-10 …

FREQ=DAILY;UNTIL=20260915
  → daily, stopping after (inclusive) 2026-09-15

FREQ=WEEKLY;BYDAY=TU;COUNT=4
  → 4 occurrences from the first Tuesday: 09-08, 09-15, 09-22, 09-29, then stops
